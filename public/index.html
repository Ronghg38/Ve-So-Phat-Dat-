const express=require("express"); const cheerio=require("cheerio"); const cron=require("node-cron"); const fs=require("fs"); const path=require("path"); const app=express(), PORT=process.env.PORT||3000; const DATA=path.join(__dirname,"data.json"); const SOURCE="https://sxmn.com.vn/"; app.use(express.json()); app.use(express.static(path.join(__dirname,"public"))); function load(){try{return JSON.parse(fs.readFileSync(DATA,"utf8"))}catch{return {updatedAt:null,source:SOURCE,days:[]}}} function save(x){fs.writeFileSync(DATA,JSON.stringify(x,null,2))} function clean(s){return (s||"").replace(/\s+/g," ").trim()} function dateVN(d=new Date()){ const z=new Date(d.getTime()+7*3600000); return z.toISOString().slice(0,10).split("-").reverse().join("/"); } async function fetchToday(){ const r=await fetch(SOURCE,{headers:{"User-Agent":"Mozilla/5.0"}}); if(!r.ok) throw new Error("Nguồn kết quả trả về HTTP "+r.status); const html=await r.text(), $=cheerio.load(html); const text=$("body").text(); // Lấy khối bảng kết quả XSMN gần nhất trên trang nguồn. const tables=[]; $("table").each((i,t)=>{ const rows=[]; $(t).find("tr").each((_,tr)=>{ const cells=$(tr).find("th,td").map((_,c)=>clean($(c).text())).get(); if(cells.length) rows.push(cells); }); if(rows.length) tables.push(rows); }); // Tìm bảng có hàng G8/G7/ĐB. const result=tables.find(t=>t.some(r=>r.some(c=>c==="G8")) && t.some(r=>r.some(c=>c==="ĐB"))); if(!result) throw new Error("Không tìm thấy bảng XSMN trên nguồn dữ liệu."); const headers=result.find(r=>r.includes("G8"))||[]; const colCount=headers.length-1; const stations=headers.slice(1); const rows={}; for(const st of stations) rows[st]={}; for(const r of result){ const label=r[0]; if(["G8","G7","G6","G5","G4","G3","G2","G1","ĐB"].includes(label)){ stations.forEach((st,i)=>{rows[st][label]=r[i+1]||""}); } } const payload={date:dateVN(),updatedAt:new Date().toISOString(),stations:rows}; const db=load(); db.days=[payload,...db.days.filter(x=>x.date!==payload.date)].slice(0,120); db.updatedAt=payload.updatedAt; save(db); return payload; } function digits(s){return (s.match(/\d/g)||[]).map(Number)} function total(s){return digits(s).reduce((a,b)=>a+b,0)%10} function loto(v){return [...new Set(v.filter(Boolean).flatMap(s=>(s.match(/\d{2}/g)||[]).map(x=>x.slice(-2))))]} function analyze(day){ const stations=day.stations||{}, per={}, all=[]; for(const [name,g] of Object.entries(stations)){ const db=g["ĐB"]||"", nums=loto(Object.values(g)); const tail=db.slice(-2), head=db.slice(0,1), t=total(db); per[name]={db,total:t,head,tail,loto:nums}; all.push(...nums); } const c={}; all.forEach(n=>c[n]=(c[n]||0)+1); const cross=Object.entries(c).filter(([,n])=>n>=2).map(([n])=>n).sort(); return {date:day.date,stations:per,cross}; } app.get("/api/status",(req,res)=>{const d=load();res.json({updatedAt:d.updatedAt,source:d.source,latest:d.days[0]||null})}); app.get("/api/results",(req,res)=>{const d=load();res.json(d.days.slice(0,30))}); app.get("/api/analyze",(req,res)=>{const d=load(); if(!d.days[0]) return res.status(404).json({error:"Chưa có dữ liệu"});res.json(analyze(d.days[0]))}); app.post("/api/update",async(req,res)=>{try{const x=await fetchToday();res.json({ok:true,data:x,analysis:analyze(x)})}catch(e){res.status(500).json({ok:false,error:e.message})}}); cron.schedule("*/30 * * * *",()=>fetchToday().catch(e=>console.error("AUTO UPDATE:",e.message)),{timezone:"Asia/Ho_Chi_Minh"}); fetchToday().catch(()=>{}); app.listen(PORT,()=>console.log(`Soi Cau XSMN: http://localhost:${PORT}`));
